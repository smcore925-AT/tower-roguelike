Approved behavior
Enter Play Mode -> Start Run -> configured starting floor -> Next Floor increments by one -> Die shows Run Over and floor reached -> Restart immediately begins a new run at the configured starting floor.

Assumptions and scope
- Unity 6 LTS with URP. Use the newest Unity 6 LTS when the owner creates the project, and record the exact version in docs/DECISIONS.md. This delivery does not change the project's Unity version.
- Unity UI (uGUI), TextMeshPro, and the new Input System. This skeleton is tested by mouse clicks; keyboard/gamepad focus handling is not implemented.
- One scene, one RunManager, one RunUI. Both controller objects remain active throughout the run.
- The floor is a counter. Floor loading, player health, enemies, saving, and progression between runs are future integrations.
- Death changes RunState; it does not globally change Time.timeScale. Future gameplay components should respect RunState and the Changed event.
- Restart resets this skeleton's run state and floor counter. It does not reload the scene or reset future gameplay objects.
- RunSettings is configuration only. RunManager owns progress and exposes read-only State and CurrentFloor properties plus a Changed event. CurrentFloor is 0 before the first run, while the HUD is hidden.
- Scripts connect button listeners automatically. Leave Inspector Button On Click() lists empty.
- No Unity-generated assets, scenes, .meta files, or ProjectSettings are supplied. Unity will generate the appropriate files as the owner performs setup.

```csharp
using UnityEngine;

namespace Tower.Core
{
    [CreateAssetMenu(fileName = "RunSettings", menuName = "Tower/Run Settings")]
    public sealed class RunSettings : ScriptableObject
    {
        [SerializeField, Min(1)] private int _startingFloor = 1;

        public int StartingFloor => Mathf.Max(1, _startingFloor);

        private void OnValidate()
        {
            _startingFloor = Mathf.Max(1, _startingFloor);
        }
    }
}
```

```csharp
using System;
using UnityEngine;

namespace Tower.Core
{
    [DisallowMultipleComponent]
    public sealed class RunManager : MonoBehaviour
    {
        [SerializeField] private RunSettings _settings;

        private RunState _state = RunState.NotStarted;
        private int _currentFloor;

        public RunState State => _state;
        public int CurrentFloor => _currentFloor;

        // Raised after both State and CurrentFloor are ready to read.
        public event Action Changed;

        [ContextMenu("Start Run")]
        public void StartRun()
        {
            if (!Application.isPlaying || !isActiveAndEnabled || _state != RunState.NotStarted)
            {
                return;
            }

            BeginRun();
        }

        [ContextMenu("Advance Floor")]
        public void AdvanceFloor()
        {
            if (!Application.isPlaying || !isActiveAndEnabled || _state != RunState.Running)
            {
                return;
            }

            if (_currentFloor == int.MaxValue)
            {
                Debug.LogWarning("The floor counter has reached its numeric limit.", this);
                return;
            }

            _currentFloor++;
            Changed?.Invoke();
        }

        [ContextMenu("Die")]
        public void Die()
        {
            if (!Application.isPlaying || !isActiveAndEnabled || _state != RunState.Running)
            {
                return;
            }

            _state = RunState.RunOver;
            Changed?.Invoke();
        }

        [ContextMenu("Restart Run")]
        public void RestartRun()
        {
            if (!Application.isPlaying || !isActiveAndEnabled || _state != RunState.RunOver)
            {
                return;
            }

            BeginRun();
        }

        private void BeginRun()
        {
            if (_settings == null)
            {
                Debug.LogError("RunManager needs a RunSettings asset. Assign Settings in the Inspector.", this);
                return;
            }

            _currentFloor = _settings.StartingFloor;
            _state = RunState.Running;
            Changed?.Invoke();
        }
    }
}
```

```csharp
using TMPro;
using Tower.Core;
using UnityEngine;
using UnityEngine.UI;

namespace Tower.UI
{
    [DisallowMultipleComponent]
    public sealed class RunUI : MonoBehaviour
    {
        [Header("Run")]
        [SerializeField] private RunManager _runManager;

        [Header("Panels")]
        [SerializeField] private GameObject _startPanel;
        [SerializeField] private GameObject _runningPanel;
        [SerializeField] private GameObject _runOverPanel;

        [Header("Text")]
        [SerializeField] private TMP_Text _floorText;
        [SerializeField] private TMP_Text _floorReachedText;
        [SerializeField] private string _floorLabel = "Floor";
        [SerializeField] private string _floorReachedLabel = "Floor reached";

        [Header("Buttons")]
        [SerializeField] private Button _startButton;
        [SerializeField] private Button _nextFloorButton;
        [SerializeField] private Button _dieButton;
        [SerializeField] private Button _restartButton;

        private void OnEnable()
        {
            if (!HasRequiredReferences())
            {
                Debug.LogError("RunUI needs its manager, three panels, two text fields, and four buttons assigned.", this);
                enabled = false;
                return;
            }

            _runManager.Changed += Refresh;
            _startButton.onClick.AddListener(StartRun);
            _nextFloorButton.onClick.AddListener(AdvanceFloor);
            _dieButton.onClick.AddListener(Die);
            _restartButton.onClick.AddListener(RestartRun);
            Refresh();
        }

        private void OnDisable()
        {
            if (_runManager != null)
            {
                _runManager.Changed -= Refresh;
            }

            if (_startButton != null)
            {
                _startButton.onClick.RemoveListener(StartRun);
            }

            if (_nextFloorButton != null)
            {
                _nextFloorButton.onClick.RemoveListener(AdvanceFloor);
            }

            if (_dieButton != null)
            {
                _dieButton.onClick.RemoveListener(Die);
            }

            if (_restartButton != null)
            {
                _restartButton.onClick.RemoveListener(RestartRun);
            }
        }

        private bool HasRequiredReferences()
        {
            return _runManager != null
                && _startPanel != null
                && _runningPanel != null
                && _runOverPanel != null
                && _floorText != null
                && _floorReachedText != null
                && _startButton != null
                && _nextFloorButton != null
                && _dieButton != null
                && _restartButton != null;
        }

        private void Refresh()
        {
            bool notStarted = _runManager.State == RunState.NotStarted;
            bool running = _runManager.State == RunState.Running;
            bool runOver = _runManager.State == RunState.RunOver;

            _floorText.text = $"{_floorLabel} {_runManager.CurrentFloor}";
            _floorReachedText.text = $"{_floorReachedLabel} {_runManager.CurrentFloor}";

            _startButton.interactable = notStarted;
            _nextFloorButton.interactable = running;
            _dieButton.interactable = running;
            _restartButton.interactable = runOver;

            _startPanel.SetActive(notStarted);
            _runningPanel.SetActive(running);
            _runOverPanel.SetActive(runOver);
        }

        private void StartRun()
        {
            _runManager.StartRun();
        }

        private void AdvanceFloor()
        {
            _runManager.AdvanceFloor();
        }

        private void Die()
        {
            _runManager.Die();
        }

        private void RestartRun()
        {
            _runManager.RestartRun();
        }
    }
}
```
